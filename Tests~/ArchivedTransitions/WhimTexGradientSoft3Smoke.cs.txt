// Historical tests for retired transitions; not a current executable test suite.
using System;
using System.Reflection;
using UnityEngine;
using DCFApixels.WhimTex;

public static class WhimTexGradientSoft3Smoke
{
    public static string DuplicateEndpoints() => CheckDuplicateEndpoints(true);
    public static string MeasureDuplicateEndpoints() => CheckDuplicateEndpoints(false);
    static string CheckDuplicateEndpoints(bool requireEqual)
    {
        int samples = 0; float maxDifference = 0, neutralDifference = 0;
        foreach (var mode in new[] {WhimTexGradientMode.Classic, WhimTexGradientMode.Linear, WhimTexGradientMode.Perceptual})
        foreach (var space in new[] {ColorSpace.Gamma, ColorSpace.Linear})
        foreach (float midpoint in new[] {.01f, .25f, .5f, .75f, .99f})
        foreach (float smooth in new[] {0f, .5f, 1f})
        for (int palette = 0; palette < 3; palette++)
        {
            Color first = palette == 0 ? Color.white : palette == 1 ? Color.red : new Color(4, -.1f, .2f);
            Color middle = palette == 0 ? Color.gray : Color.cyan;
            Color last = palette == 0 ? Color.black : new Color(.2f, .1f, .8f);
            var g = new WhimTexGradient {Mode = mode, ColorSpace = space,
                Transition = WhimTexGradientTransition.Soft3, Smoothness = smooth};
            g.SetKeys(new[] {new GradientColorKey(first, .2f), new GradientColorKey(middle, .5f), new GradientColorKey(last, .8f)},
                new[] {new GradientAlphaKey(.1f, .22f), new GradientAlphaKey(.8f, .57f), new GradientAlphaKey(.3f, .78f)});
            g.SetMidpoint(false, 0, midpoint); g.SetMidpoint(false, 1, Mathf.Clamp(1-midpoint, .01f, .99f));
            g.SetMidpoint(true, 0, midpoint); g.SetMidpoint(true, 1, Mathf.Clamp(1-midpoint, .01f, .99f));
            var standard = g.Clone(); standard.Transition = WhimTexGradientTransition.Standard;
            for (int sides = 1; sides <= 3; sides++)
            {
                var expanded = g.Clone();
                var colors = new System.Collections.Generic.List<GradientColorKey>(g.ColorKeys);
                var alphas = new System.Collections.Generic.List<GradientAlphaKey>(g.AlphaKeys);
                if ((sides & 1) != 0) { colors.Insert(0, new GradientColorKey(first, .05f)); alphas.Insert(0, new GradientAlphaKey(.1f, .05f)); }
                if ((sides & 2) != 0) { colors.Add(new GradientColorKey(last, .95f)); alphas.Add(new GradientAlphaKey(.3f, .95f)); }
                expanded.SetKeys(colors.ToArray(), alphas.ToArray());
                for (int i = 0; i <= 1000; i++)
                {
                    float t = i/1000f; var a = g.Evaluate(t); var b = expanded.Evaluate(t);
                    for (int c = 0; c < 4; c++)
                    {
                        float difference = Mathf.Abs(a[c]-b[c])/Mathf.Max(1, Mathf.Abs(a[c]));
                        maxDifference = Mathf.Max(maxDifference, difference); samples++;
                    }
                    if (palette == 0)
                    {
                        var s = standard.Evaluate(t);
                        for (int c = 0; c < 4; c++) neutralDifference = Mathf.Max(neutralDifference, Mathf.Abs(a[c]-s[c]));
                    }
                }
            }
        }
        if (requireEqual && (maxDifference > .000003f || neutralDifference > .000003f))
            throw new Exception("Constant continuation changed ramp: " + maxDifference + "; neutral Standard difference: " + neutralDifference);
        return "Duplicate endpoint samples=" + samples + "; max relative difference=" + maxDifference + "; neutral Standard difference=" + neutralDifference;
    }
    static int checks;
    static void Check(bool condition, string message) { checks++; if (!condition) throw new Exception(message); }
    static void Near(Color a, Color b, float tolerance = .001f)
    {
        for (int c = 0; c < 4; c++) Check(Mathf.Abs(a[c]-b[c]) <= tolerance*Mathf.Max(1, Mathf.Abs(b[c])), "Color mismatch: " + a + " / " + b);
    }
    public static string Main()
    {
        checks = 0;
        Check((int)WhimTexGradientTransition.Standard == 0 && (int)WhimTexGradientTransition.Soft == 1 &&
            (int)WhimTexGradientTransition.Soft2 == 2 && (int)WhimTexGradientTransition.Soft3 == 3, "Serialized enum IDs");
        var g = new WhimTexGradient { Mode = WhimTexGradientMode.Perceptual, Transition = WhimTexGradientTransition.Soft3 };
        var reductionField = typeof(WhimTexGradient).GetField("chromaReduction", BindingFlags.Instance | BindingFlags.NonPublic);
        var opaque = new[] {new GradientAlphaKey(1, 0), new GradientAlphaKey(1, 1)};
        void Pair(Color a, Color b)
        {
            g.SetKeys(new[] {new GradientColorKey(a, 0), new GradientColorKey(b, 1)}, opaque);
            g.Evaluate(.5f);
        }
        Pair(Color.white, Color.red);
        Check(((float[])reductionField.GetValue(g))[0] == 0, "Neutral endpoint must disable chroma correction");
        g.ColorSpace = ColorSpace.Linear;
        Pair(new Color(.8f, .04f, .02f), new Color(.2f, .01f, .005f));
        Check(((float[])reductionField.GetValue(g))[0] < .000001f, "Same hue must retain chroma");
        Pair(Color.red, Color.cyan);
        Check(((float[])reductionField.GetValue(g))[0] > .1f, "Opposing hues should enable correction");
        foreach (var space in new[] {ColorSpace.Gamma, ColorSpace.Linear})
        {
            g.ColorSpace = space;
            g.SetKeys(new[] {new GradientColorKey(Color.red, .2f), new GradientColorKey(Color.cyan, .8f)}, opaque);
            Near(g.Evaluate(.20001f), g.Evaluate(.2f), .000001f);
            Near(g.Evaluate(.79999f), g.Evaluate(.8f), .000001f);
        }
        g.Mode = WhimTexGradientMode.Linear; g.Evaluate(.3f);
        Check(reductionField.GetValue(g) == null, "Mode change must clear correction cache");
        g.Mode = WhimTexGradientMode.Perceptual;
        g.Transition = WhimTexGradientTransition.Soft2; g.Evaluate(.3f);
        Check(reductionField.GetValue(g) == null, "Transition change must clear correction cache");
        g.Transition = WhimTexGradientTransition.Soft3;
        Pair(Color.red, Color.red);
        for (int i = 0; i <= 100; i++) Near(g.Evaluate(i/100f), Color.red);

        // The scalar basis uses the same constant continuation at outer and inner stops.
        var ramp = new WhimTexGradient {Transition = WhimTexGradientTransition.Soft3};
        var standardRamp = ramp.Clone(); standardRamp.Transition = WhimTexGradientTransition.Standard;
        for (int i = 0; i <= 1000; i++)
        {
            Check(ramp.Evaluate(i/1000f).Equals(standardRamp.Evaluate(i/1000f)), "Scalar basis differs from constant continuation");
        }

        var random = new System.Random(7931);
        for (int trial = 0; trial < 80; trial++)
        {
            g.ColorSpace = trial % 2 == 0 ? ColorSpace.Gamma : ColorSpace.Linear;
            var keys = new GradientColorKey[4];
            for (int i = 0; i < keys.Length; i++)
            {
                Color c = new Color((float)random.NextDouble(), (float)random.NextDouble(), (float)random.NextDouble());
                if (trial % 3 == 0) c = c*8 - new Color(1, 1, 1, 0);
                c.a = 1; keys[i] = new GradientColorKey(c, .05f + i*.3f);
            }
            g.SetKeys(keys, opaque);
            for (int i = 0; i < 3; i++) g.SetMidpoint(false, i, .05f+(float)random.NextDouble()*.9f);
            var reversed = g.Clone();
            var reversedKeys = new GradientColorKey[4];
            for (int i = 0; i < 4; i++) reversedKeys[i] = new GradientColorKey(keys[3-i].color, 1-keys[3-i].time);
            reversed.SetKeys(reversedKeys, opaque);
            for (int i = 0; i < 3; i++) reversed.SetMidpoint(false, i, 1-g.GetMidpoint(false, 2-i));
            g.Smoothness = reversed.Smoothness = 1;
            foreach (var key in keys) Near(g.Evaluate(key.time), key.color, .002f);
            for (int i = 0; i <= 200; i++)
            {
                float t = i/200f; Color value = g.Evaluate(t);
                for (int c = 0; c < 4; c++) Check(!float.IsNaN(value[c]) && !float.IsInfinity(value[c]), "Signed HDR finite");
                Near(value, reversed.Evaluate(1-t), .003f);
            }
            var standard = g.Clone(); standard.Transition = WhimTexGradientTransition.Standard;
            g.Smoothness = standard.Smoothness = 0;
            for (int i = 0; i <= 100; i++) Check(g.Evaluate(i/100f).Equals(standard.Evaluate(i/100f)), "Zero smoothness must be bit-identical");
        }
        g.Smoothness = 1; g.Evaluate(.3f);
        long before = GC.GetAllocatedBytesForCurrentThread(); float sink = 0;
        for (int i = 0; i < 10000; i++) sink += g.Evaluate(i/10000f).r;
        Check(GC.GetAllocatedBytesForCurrentThread() == before && !float.IsNaN(sink), "Perceptual hot path allocated");
        var independent = g.Clone();
        independent.SetKeys(new[] {new GradientColorKey(Color.black, 0), new GradientColorKey(Color.white, 1)}, g.AlphaKeys);
        for (int i = 0; i <= 200; i++) Check(g.Evaluate(i/200f).a == independent.Evaluate(i/200f).a, "Color correction changed alpha");
        RenderEndpoints();
        return "Soft 3 color contracts: " + checks + " checks passed. Endpoint comparison: Temp/WhimTex/soft3-endpoints.png";
    }
    static void RenderEndpoints()
    {
        var g = new WhimTexGradient {Transition = WhimTexGradientTransition.Soft3};
        g.SetKeys(new[] {new GradientColorKey(Color.white, .35f), new GradientColorKey(Color.black, .95f)},
            new[] {new GradientAlphaKey(1, 0), new GradientAlphaKey(1, 1)});
        var image = new Texture2D(512, 256, TextureFormat.RGBA32, false, true);
        try
        {
            var pixels = new Color[512*256];
            for (int y = 0; y < 256; y++) for (int x = 0; x < 256; x++)
            {
                float radius = new Vector2((x+.5f-128)/120, (y+.5f-128)/120).magnitude;
                float t = Mathf.Clamp01((radius-.35f)/.6f), u = t <= .5f ? 2*t : 2*(1-t);
                // Rejected local endpoint remap, kept only as a visual regression baseline.
                if (u < .25f) { float v = u/.25f; u = .25f*v*v*v*(6+v*(-8+3*v)); }
                float old = .5f*((-2*u*u*u+3*u*u) + .75f*(u*u*u-2*u*u+u) + (u*u*u-u*u));
                old = t <= .5f ? 1-old : old;
                pixels[y*512+x] = new Color(old, old, old, 1);
                pixels[y*512+x+256] = g.Evaluate(radius);
            }
            image.SetPixels(pixels); image.Apply();
            System.IO.Directory.CreateDirectory("Temp/WhimTex");
            System.IO.File.WriteAllBytes("Temp/WhimTex/soft3-endpoints.png", image.EncodeToPNG());
        }
        finally { UnityEngine.Object.DestroyImmediate(image); }
    }
}
