// Historical tests for retired transitions; not a current executable test suite.
using System;
using System.IO;
using UnityEngine;
using DCFApixels.WhimTex;

public static class WhimTexGradientSoftJoinSmoke
{
    static int checks;
    static void Check(bool condition, string message)
    {
        checks++;
        if (!condition) throw new Exception(message);
    }

    static WhimTexGradient Ramp(float[] values)
    {
        var colors = new GradientColorKey[values.Length];
        var alpha = new GradientAlphaKey[values.Length];
        for (int i = 0; i < values.Length; i++)
        {
            float t = i / (float)(values.Length - 1), v = values[i];
            colors[i] = new GradientColorKey(new Color(v, v, v, 1), t);
            alpha[i] = new GradientAlphaKey(Mathf.Clamp01(v), t);
        }
        var g = new WhimTexGradient
        {
            Mode = WhimTexGradientMode.Classic, ColorSpace = ColorSpace.Gamma,
            Transition = WhimTexGradientTransition.Soft, Smoothness = 1
        };
        g.SetKeys(colors, alpha);
        return g;
    }

    public static string Main()
    {
        checks = 0;
        var g = Ramp(new[] {0f, .25f, .5f, .75f, 1f});
        // Independent easing per real-key interval fails both of these regressions.
        for (int i = 125; i <= 875; i++)
        {
            float t = i / 1000f;
            var c = g.Evaluate(t);
            Check(Mathf.Abs(c.r-t) < .000002f && Mathf.Abs(c.a-t) < .000002f,
                "Collinear interior ramp must not acquire waves");
        }
        foreach (float t in new[] {.25f, .5f, .75f})
        {
            const float e = .001f;
            var c = g.Evaluate(t); var left = g.Evaluate(t-e); var right = g.Evaluate(t+e);
            for (int channel = 0; channel < 4; channel++)
            {
                float dl = (c[channel]-left[channel])/e, dr = (right[channel]-c[channel])/e;
                Check(dl > .99f && dr > .99f && Mathf.Abs(dl-dr) < .001f,
                    "Monotone real key must keep a shared nonzero slope");
            }
        }
        g = Ramp(new[] {0f, .15f, .7f, .85f, 1f});
        for (int i = 0; i < 4; i++)
        {
            g.SetMidpoint(false, i, i % 2 == 0 ? .3f : .7f);
            g.SetMidpoint(true, i, i % 2 == 0 ? .3f : .7f);
        }
        foreach (float t in new[] {.075f, .25f, .425f, .5f, .575f, .75f, .925f})
        {
            const float e = .0005f;
            float c = g.Evaluate(t).r;
            float dl = (c-g.Evaluate(t-e).r)/e, dr = (g.Evaluate(t+e).r-c)/e;
            Check(dl > .05f && dr > .05f && Mathf.Abs(dl-dr) < .006f,
                "Shared slope with asymmetric neighboring midpoints at " + t);
            // Estimate curvature at the join, not at t +/- e.
            const float h = .002f;
            double ddl = (35.0*c-104.0*g.Evaluate(t-h).r+114.0*g.Evaluate(t-2*h).r-56.0*g.Evaluate(t-3*h).r+11.0*g.Evaluate(t-4*h).r)/(12*h*h);
            double ddr = (35.0*c-104.0*g.Evaluate(t+h).r+114.0*g.Evaluate(t+2*h).r-56.0*g.Evaluate(t+3*h).r+11.0*g.Evaluate(t+4*h).r)/(12*h*h);
            Check(Math.Abs(ddl) < 1 && Math.Abs(ddr) < 1, "Curvature join at " + t + ": " + ddl + ", " + ddr);
        }
        var random = new System.Random(5137);
        for (int run = 0; run < 64; run++)
        {
            var values = new float[9];
            for (int i = 0; i < values.Length; i++) values[i] = (float)random.NextDouble();
            Array.Sort(values);
            for (int i = 0; i < values.Length; i++) values[i] = values[i]*6-2;
            if (run % 2 != 0) Array.Reverse(values);
            g = Ramp(values);
            for (int i = 0; i < values.Length-1; i++)
            {
                float midpoint = run % 3 == 0 ? (i % 2 == 0 ? .01f : .99f) : .01f+.98f*(float)random.NextDouble();
                g.SetMidpoint(false, i, midpoint);
                float t = (i+midpoint)/(values.Length-1);
                Check(Mathf.Abs(g.Evaluate(t).r-(values[i]+values[i+1])*.5f) < .0001f,
                    "Midpoint value constraint");
            }
            float previous = g.Evaluate(0).r;
            for (int i = 0; i <= 4096; i++)
            {
                float v = g.Evaluate(i/4096f).r;
                Check(!float.IsNaN(v) && !float.IsInfinity(v) && v >= -2.00001f && v <= 4.00001f, "Finite signed HDR");
                Check(run % 2 == 0 ? v >= previous-.000002f : v <= previous+.000002f, "Random monotonicity");
                previous = v;
            }
        }
        g = Ramp(new[] {0f, .5f, 1f, .5f, 0f});
        Check(Mathf.Abs(g.Evaluate(.4999f).r-1) < .000001f, "Extremum remains smooth");
        g = Ramp(new[] {0f, .5f, .5f, .5f, 1f});
        for (int i = 250; i <= 750; i++) Check(g.Evaluate(i/1000f).r == .5f, "Authored plateau preserved");

        // Diagnostic only, outside Assets: Standard | connected Soft | old per-span easing.
        var image = new Texture2D(450, 512, TextureFormat.RGBA32, false, true);
        try
        {
            float[] values = {0f, .15f, .45f, .8f, 1f, .6f, .1f, 0f};
            g = Ramp(values); var standard = g.Clone(); standard.Transition = WhimTexGradientTransition.Standard;
            for (int y = 0; y < image.height; y++)
            {
                float t = 1-y/(float)(image.height-1), p = t*(values.Length-1);
                int segment = Mathf.Min((int)p, values.Length-2); float u = p-segment;
                float old = Mathf.Lerp(values[segment], values[segment+1], u*u*u*(u*(u*6-15)+10));
                for (int x = 0; x < image.width; x++)
                {
                    float v = x < 150 ? standard.Evaluate(t).r : x < 300 ? g.Evaluate(t).r : old;
                    image.SetPixel(x, y, x % 150 < 3 ? Color.gray : new Color(v, v, v, 1));
                }
            }
            image.Apply(); Directory.CreateDirectory("Temp/WhimTex");
            File.WriteAllBytes("Temp/WhimTex/soft-join-comparison.png", image.EncodeToPNG());
        }
        finally { UnityEngine.Object.DestroyImmediate(image); }
        return "Connected Soft: " + checks + " checks passed; comparison: Temp/WhimTex/soft-join-comparison.png";
    }
}
